#include <Wire.h>
#include "DFRobot_OxygenSensor.h"

#define O2_I2C_ADDRESS 0x73

DFRobot_OxygenSensor oxygen;

void setup() {
  Serial.begin(9600);
  Wire.begin();

  while (!oxygen.begin(O2_I2C_ADDRESS)) {
    Serial.println("Oxygen sensor not detected!");
    delay(1000);
  }

  Serial.println("Oxygen sensor ready");
}

void loop() {
  float o2 = oxygen.getOxygenData(10); // average of 10 samples

  Serial.print("Oxygen concentration: ");
  Serial.print(o2, 2);
  Serial.println(" %");

  delay(1000);
}
